## 一、核心实现

### 1.1 EventBus 类（TypeScript）

```typescript
// src/utils/eventBus.ts

type EventCallback<T = any> = (data: T) => void;

class EventBus {
  private events: Map<string, Set<EventCallback>> = new Map();
  private history: Array<{ event: string; data: any; timestamp: number }> = [];
  private maxHistory: number = 100;

  // 订阅事件
  on<T>(event: string, callback: EventCallback<T>): () => void {
    if (!this.events.has(event)) {
      this.events.set(event, new Set());
    }
    this.events.get(event)!.add(callback);

    // 返回取消订阅函数
    return () => {
      this.events.get(event)?.delete(callback);
    };
  }

  // 订阅一次性事件
  once<T>(event: string, callback: EventCallback<T>): void {
    const wrapper = (data: T) => {
      callback(data);
      this.off(event, wrapper);
    };
    this.on(event, wrapper);
  }

  // 取消订阅
  off<T>(event: string, callback: EventCallback<T>): void {
    this.events.get(event)?.delete(callback);
  }

  // 发送事件
  emit<T>(event: string, data?: T): void {
    // 记录历史（用于调试）
    this.history.push({ event, data, timestamp: Date.now() });
    if (this.history.length > this.maxHistory) {
      this.history.shift();
    }

    // 执行回调
    const callbacks = this.events.get(event);
    if (callbacks) {
      callbacks.forEach((callback) => {
        try {
          callback(data);
        } catch (error) {
          console.error(`[EventBus] Error in event "${event}":`, error);
        }
      });
    }

    // 开发环境日志
    if (import.meta.env.DEV) {
      console.log(`[EventBus] ${event}`, data);
    }
  }

  // 获取事件历史（调试用）
  getHistory(): Array<{ event: string; data: any; timestamp: number }> {
    return [...this.history];
  }

  // 清空历史
  clearHistory(): void {
    this.history = [];
  }

  // 获取当前订阅数（调试用）
  getSubscriberCount(event?: string): number {
    if (event) {
      return this.events.get(event)?.size || 0;
    }
    return Array.from(this.events.values()).reduce(
      (sum, set) => sum + set.size, 0
    );
  }
}

// 单例导出
export const eventBus = new EventBus();
export type { EventCallback };
```

### 1.2 React Hook 封装

```typescript
// src/hooks/useEventBus.ts

import { useEffect, useCallback } from 'react';
import { eventBus, EventCallback } from '@/utils/eventBus';

export function useEventBus<T = any>(
  event: string,
  callback: EventCallback<T>,
  deps: React.DependencyList = []
) {
  useEffect(() => {
    const unsubscribe = eventBus.on<T>(event, callback);
    return unsubscribe;
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [event, ...deps]);
}

export function useEmit() {
  return useCallback(<T = any>(event: string, data?: T) => {
    eventBus.emit(event, data);
  }, []);
}
```

## 三、模块间通信示例

### 3.1 场景：用户说"新建一个工作会话"

```typescript
// ===== 语音系统（发送方）=====
// src/voice/hooks/useVoiceChat.ts

import { eventBus } from '@/utils/eventBus';

async function handleUserMessage(text: string, sessionId: string) {
  // 1. 发送消息给后端
  const response = await sendMessageToBackend(text, sessionId);

  // 2. 检查是否识别到"新建会话"意图
  if (response.intent?.type === 'new_session') {
    // 3. 通知宇宙系统：准备生长动画
    eventBus.emit('voice:session:create', {
      title: response.intent.params.title || '新会话',
      topic: response.intent.params.topic || 'other',
      parentId: sessionId !== 'home' ? sessionId : undefined,
    });

    // 4. 等待后端创建完成
    const newSession = await createSessionAPI({
      title: response.intent.params.title,
      topic: response.intent.params.topic,
      parentId: sessionId !== 'home' ? sessionId : undefined,
    });

    // 5. 通知宇宙系统：新星球可以生成了
    eventBus.emit('universe:planet:spawned', {
      sessionId: newSession.id,
      title: newSession.title,
      topic: newSession.topic,
      position: [0, 0, 0], // 宇宙系统会计算实际位置
      parentId: newSession.parentId,
    });

    // 6. 切换到新会话
    eventBus.emit('voice:session:switch', {
      sessionId: newSession.id,
    });
  }
}

// ===== 宇宙系统（接收方）=====
// src/universe/components/Universe.tsx

import { useEventBus } from '@/hooks/useEventBus';

function Universe() {
  const [planets, setPlanets] = useState<Planet[]>([]);

  // 监听：语音系统请求创建新会话
  useEventBus('voice:session:create', (data) => {
    console.log('[Universe] 准备生长新星球:', data);
    // 预计算位置，准备动画
  });

  // 监听：后端确认创建完成，开始生长动画
  useEventBus('universe:planet:spawned', (data) => {
    console.log('[Universe] 开始生长动画:', data);

    // 1. 找到源星球位置
    const sourcePlanet = data.parentId 
      ? planets.find(p => p.id === data.parentId)
      : { position: new Vector3(0, 0, 0) }; // 主星球

    // 2. 计算新星球位置（随机方向，15-25单位距离）
    const direction = new Vector3(
      Math.random() - 0.5,
      Math.random() - 0.5,
      Math.random() - 0.5
    ).normalize();
    const distance = 15 + Math.random() * 10;
    const newPosition = sourcePlanet.position.clone().add(
      direction.multiplyScalar(distance)
    );

    // 3. 触发分裂光束动画
    eventBus.emit('universe:effect:spawnBeam', {
      from: sourcePlanet.position.toArray(),
      to: newPosition.toArray(),
      color: getTopicColor(data.topic),
    });

    // 4. 光束动画完成后，添加新星球
    setTimeout(() => {
      setPlanets(prev => [...prev, {
        id: data.sessionId,
        position: newPosition,
        // ... 其他属性
      }]);

      // 5. 相机飞行到新星球
      eventBus.emit('universe:camera:startFly', {
        target: data.sessionId,
      });
    }, 800); // 0.8s 光束飞行时间
  });

  // 监听：相机飞行完成
  useEventBus('universe:camera:arrived', (data) => {
    console.log('[Universe] 相机已到达:', data.target);
  });

  return (
    <Canvas>
      {/* 3D 场景 */}
    </Canvas>
  );
}
```

```typescript
// src/universe/components/Planet.tsx

function Planet({ data }: { data: PlanetData }) {
  const meshRef = useRef<Mesh>(null);

  const handleClick = () => {
    // 1. 发送点击事件
    eventBus.emit('universe:planet:click', {
      sessionId: data.sessionId,
      title: data.title,
      position: data.position.toArray(),
    });

    // 2. 相机飞行到该星球
    eventBus.emit('universe:camera:startFly', {
      target: data.sessionId,
    });
  };

  const handleDoubleClick = () => {
    eventBus.emit('universe:planet:doubleclick', {
      sessionId: data.sessionId,
      title: data.title,
    });
  };

  return (
    <mesh
      ref={meshRef}
      position={data.position}
      onClick={handleClick}
      onDoubleClick={handleDoubleClick}
    >
      <sphereGeometry args={[data.radius, 32, 32]} />
      <meshStandardMaterial color={data.color} />
    </mesh>
  );
}
```

```typescript
// src/utils/eventBusDebugger.ts

import { eventBus } from './eventBus';

export function initEventBusDebugger() {
  if (!import.meta.env.DEV) return;

  // 在 window 上挂载调试方法
  (window as any).eventBus = {
    // 查看所有订阅
    subscribers: () => {
      const events = eventBus['events']; // 访问私有属性（仅调试）
      const result: Record<string, number> = {};
      events.forEach((callbacks, event) => {
        result[event] = callbacks.size;
      });
      console.table(result);
      return result;
    },

    // 查看最近事件历史
    history: () => {
      const history = eventBus.getHistory();
      console.table(history.slice(-20)); // 最近20条

    // 模拟发送事件

      eventBus.emit(event, data);
      console.log(`[Debug] Manually emitted: ${event}`, data);
    },

    // 清除历史
    clear: () => {
      eventBus.clearHistory();
      console.log('[Debug] Event history cleared');
    },
  };

  console.log('[EventBus Debugger] Initialized. Use window.eventBus to debug.');
}
```

## 五、使用规范

### 5.1 命名规范

```
模块名:事件类型:动作

示例：
- universe:planet:click      宇宙系统 - 星球 - 点击
- voice:recording:start      语音系统 - 录音 - 开始
- voice:session:create       语音系统 - 会话 - 创建
- nav:search:select          导航系统 - 搜索 - 选择
- memory:saved               记忆系统 - 已保存
- app:ready                  全局 - 应用就绪
```

### 5.2 最佳实践

1. **事件数据尽量轻量**：不要传递大型对象，传递 ID 或关键字段
2. **及时取消订阅**：组件卸载时取消订阅，防止内存泄漏
3. **不要循环触发**：A 触发 B，B 不要直接触发 A，避免死循环
4. **使用 TypeScript**：严格定义事件类型，编译时检查
5. **错误处理**：事件回调中捕获错误，不要影响其他监听者
