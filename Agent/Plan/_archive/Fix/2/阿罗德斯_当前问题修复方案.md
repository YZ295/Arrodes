## 问题 1：星球太丑，不圆润

**现状**：星球是多边形平面拼接的，看起来像低模游戏

**原因**：`SphereGeometry` 的分段数（segments）太低，默认是 8 或 16

**修复**：增加分段数 + 添加平滑材质 + 光照

```tsx
// HomePlanet.tsx 修复
import { useRef } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';

export function HomePlanet() {
  const meshRef = useRef<THREE.Mesh>(null);
  const glowRef = useRef<THREE.Mesh>(null);

  // 脉动动画
  useFrame((state) => {
    if (meshRef.current) {
      const scale = 1 + Math.sin(state.clock.elapsedTime * 2) * 0.05;
      meshRef.current.scale.setScalar(scale);
    }
    if (glowRef.current) {
      const opacity = 0.3 + Math.sin(state.clock.elapsedTime * 2) * 0.2;
      (glowRef.current.material as THREE.MeshBasicMaterial).opacity = opacity;
    }
  });

  return (
    <group>
      {/* 主星球 - 高分段数 + 金属材质 */}
      <mesh ref={meshRef}>
        <sphereGeometry args={[3, 64, 64]} />  {/* ✅ 64x64 分段，非常圆润 */}
        <meshStandardMaterial
          color="#FFD700"
          metalness={0.8}
          roughness={0.2}
          emissive="#FFD700"
          emissiveIntensity={0.3}
        />
      </mesh>

      {/* 内层光晕 */}
      <mesh ref={glowRef} scale={1.2}>
        <sphereGeometry args={[3, 32, 32]} />
        <meshBasicMaterial
          color="#FFD700"
          transparent
          opacity={0.3}
          side={THREE.BackSide}
        />
      </mesh>

      {/* 外层光晕（更大更淡） */}
      <mesh scale={1.5}>
        <sphereGeometry args={[3, 32, 32]} />
        <meshBasicMaterial
          color="#FFD700"
          transparent
          opacity={0.1}
          side={THREE.BackSide}
        />
      </mesh>

      {/* 环形光晕（水平环） */}
      <mesh rotation={[Math.PI / 2, 0, 0]}>
        <torusGeometry args={[4.5, 0.05, 16, 100]} />
        <meshBasicMaterial color="#FFD700" transparent opacity={0.4} />
      </mesh>

      {/* 垂直环（交叉） */}
      <mesh rotation={[0, 0, Math.PI / 2]}>
        <torusGeometry args={[4.5, 0.03, 16, 100]} />
        <meshBasicMaterial color="#FFD700" transparent opacity={0.2} />
      </mesh>
    </group>
  );
}
```

```tsx
// Planet.tsx 修复（普通星球）
import { useRef, useState } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';
import { Text } from '@react-three/drei';
import { TOPIC_COLORS, type Topic } from '@shared/types';

interface PlanetProps {
  data: {
    id: string;
    sessionId: string;
    position: [number, number, number];
    radius: number;
    topic: Topic;
    title: string;
    messageCount: number;
    lastActiveAt: string;
  };
  isSelected?: boolean;
  onClick?: () => void;
}

export function Planet({ data, isSelected, onClick }: PlanetProps) {
  const meshRef = useRef<THREE.Mesh>(null);
  const [hovered, setHovered] = useState(false);

  const color = TOPIC_COLORS[data.topic] || '#6B7280';
  const glowIntensity = isSelected ? 0.8 : hovered ? 0.5 : 0.2;

  useFrame((state) => {
    if (meshRef.current) {
      // 自转
      meshRef.current.rotation.y += 0.005;

      // 选中时脉冲
      if (isSelected) {
        const scale = 1 + Math.sin(state.clock.elapsedTime * 3) * 0.1;
        meshRef.current.scale.setScalar(scale);
      } else {
        meshRef.current.scale.setScalar(1);
      }
    }
  });

  return (
    <group position={data.position}>
      {/* 星球本体 - 高圆润度 */}
      <mesh
        ref={meshRef}
        onClick={(e) => {
          e.stopPropagation();
          onClick?.();
        }}
        onPointerOver={() => setHovered(true)}
        onPointerOut={() => setHovered(false)}
      >
        <sphereGeometry args={[data.radius, 64, 64]} />  {/* ✅ 64x64 */}
        <meshStandardMaterial
          color={color}
          metalness={0.6}
          roughness={0.3}
          emissive={color}
          emissiveIntensity={glowIntensity}
        />
      </mesh>

      {/* 光晕 */}
      <mesh scale={1.3}>
        <sphereGeometry args={[data.radius, 32, 32]} />
        <meshBasicMaterial
          color={color}
          transparent
          opacity={glowIntensity * 0.3}
          side={THREE.BackSide}
        />
      </mesh>

      {/* 标签 - 始终面向相机 */}
      <Text
        position={[0, -data.radius - 0.8, 0]}
        fontSize={0.5}
        color="white"
        anchorX="center"
        anchorY="top"
        outlineWidth={0.02}
        outlineColor="black"
      >
        {data.title}
      </Text>
    </group>
  );
}
```

**关键修改**：
- `sphereGeometry args={[radius, 64, 64]}` — 分段数从默认 16 提升到 64
- `meshStandardMaterial` — 使用标准材质（需要光照）
- `metalness + roughness` — 金属质感
- `emissive` — 自发光，让星球有辉光感
- 多层光晕 — 内层亮、外层淡，营造氛围

## 问题 2：语音功能用不了

**可能原因**：
1. 浏览器没给麦克风权限
2. 代码里没处理权限请求
3. `useAudioRecorder` 有 bug

**修复**：添加权限请求 + 错误提示

```tsx
// src/voice/hooks/useAudioRecorder.ts

import { useState, useRef, useCallback } from 'react';

interface UseAudioRecorderReturn {
  isRecording: boolean;
  startRecording: () => Promise<void>;
  stopRecording: () => Blob | null;
  error: string | null;
  duration: number;
  volume: number;
}

export function useAudioRecorder(): UseAudioRecorderReturn {
  const [isRecording, setIsRecording] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [duration, setDuration] = useState(0);
  const [volume, setVolume] = useState(0);

  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const audioChunksRef = useRef<Blob[]>([]);
  const streamRef = useRef<MediaStream | null>(null);
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const analyserRef = useRef<AnalyserNode | null>(null);

  const startRecording = useCallback(async () => {
    try {
      setError(null);
      audioChunksRef.current = [];

      // 1. 请求麦克风权限
      const stream = await navigator.mediaDevices.getUserMedia({
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          sampleRate: 16000,
        }
      });
      streamRef.current = stream;

      // 2. 设置音量检测
      const audioContext = new AudioContext();
      const source = audioContext.createMediaStreamSource(stream);
      const analyser = audioContext.createAnalyser();
      analyser.fftSize = 256;
      source.connect(analyser);
      analyserRef.current = analyser;

      // 3. 创建 MediaRecorder
      const mediaRecorder = new MediaRecorder(stream, {
        mimeType: 'audio/webm;codecs=opus'
      });
      mediaRecorderRef.current = mediaRecorder;

      mediaRecorder.ondataavailable = (event) => {
        if (event.data.size > 0) {
          audioChunksRef.current.push(event.data);
        }
      };

      mediaRecorder.start(100); // 每 100ms 收集一次
      setIsRecording(true);

      // 4. 开始计时
      const startTime = Date.now();
      intervalRef.current = setInterval(() => {
        setDuration(Math.floor((Date.now() - startTime) / 1000));

        // 检测音量
        if (analyserRef.current) {
          const dataArray = new Uint8Array(analyserRef.current.frequencyBinCount);
          analyserRef.current.getByteFrequencyData(dataArray);
          const average = dataArray.reduce((a, b) => a + b) / dataArray.length;
          setVolume(average / 255); // 0-1
        }
      }, 100);

    } catch (err) {
      console.error('[AudioRecorder] Failed to start:', err);

      let errorMessage = '无法启动录音';
      if (err instanceof DOMException) {
        if (err.name === 'NotAllowedError') {
          errorMessage = '麦克风权限被拒绝，请在浏览器设置中允许麦克风访问';
        } else if (err.name === 'NotFoundError') {
          errorMessage = '未找到麦克风设备';
        } else if (err.name === 'NotSupportedError') {
          errorMessage = '浏览器不支持录音功能';
        }
      }

      setError(errorMessage);
      setIsRecording(false);
    }
  }, []);

  const stopRecording = useCallback((): Blob | null => {
    if (intervalRef.current) {
      clearInterval(intervalRef.current);
      intervalRef.current = null;
    }

    if (mediaRecorderRef.current && isRecording) {
      mediaRecorderRef.current.stop();

      // 停止所有轨道
      streamRef.current?.getTracks().forEach(track => track.stop());

      setIsRecording(false);
      setDuration(0);
      setVolume(0);

      // 合并音频块
      const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
      return audioBlob;
    }

    return null;
  }, [isRecording]);

  return {
    isRecording,
    startRecording,
    stopRecording,
    error,
    duration,
    volume,
  };
}
```

## 问题 3：圆形粒子音波

**效果**：录音时显示一圈粒子围绕麦克风按钮跳动

```tsx
// src/voice/components/VoiceWaveform.tsx

import { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';

interface VoiceWaveformProps {
  volume: number; // 0-1
  color?: string;
}

export function VoiceWaveform({ volume, color = '#EF4444' }: VoiceWaveformProps) {
  const groupRef = useRef<THREE.Group>(null);
  const particleCount = 60;

  const { positions, initialPositions } = useMemo(() => {
    const positions = new Float32Array(particleCount * 3);
    const initialPositions = new Float32Array(particleCount * 3);

    for (let i = 0; i < particleCount; i++) {
      const angle = (i / particleCount) * Math.PI * 2;
      const radius = 1.5; // 基础半径

      const x = Math.cos(angle) * radius;
      const y = Math.sin(angle) * radius;
      const z = 0;

      positions[i * 3] = x;
      positions[i * 3 + 1] = y;
      positions[i * 3 + 2] = z;

      initialPositions[i * 3] = x;
      initialPositions[i * 3 + 1] = y;
      initialPositions[i * 3 + 2] = z;
    }

    return { positions, initialPositions };
  }, []);

  useFrame(() => {
    if (!groupRef.current) return;

    const positionAttr = groupRef.current.children[0].geometry.attributes.position;
    const posArray = positionAttr.array as Float32Array;

    for (let i = 0; i < particleCount; i++) {
      const angle = (i / particleCount) * Math.PI * 2;

      // 音量越大，粒子跳得越远
      const jump = volume * 1.5; // 最大跳跃距离
      const noise = Math.sin(Date.now() * 0.01 + i) * 0.2; // 微小随机抖动

      const radius = 1.5 + jump + noise;

      posArray[i * 3] = Math.cos(angle) * radius;
      posArray[i * 3 + 1] = Math.sin(angle) * radius;
      posArray[i * 3 + 2] = volume * 0.5; // z轴也随音量变化
    }

    positionAttr.needsUpdate = true;
  });

  const colorObj = useMemo(() => new THREE.Color(color), [color]);

  return (
    <group ref={groupRef}>
      <points>
        <bufferGeometry>
          <bufferAttribute
            attach="attributes-position"
            args={[positions, 3]}
          />
        </bufferGeometry>
        <pointsMaterial
          size={0.15}
          color={colorObj}
          transparent
          opacity={0.8 + volume * 0.2}
          sizeAttenuation
          blending={THREE.AdditiveBlending}
          depthWrite={false}
        />
      </points>

      {/* 中心圆环 */}
      <mesh>
        <ringGeometry args={[1.3, 1.4, 64]} />
        <meshBasicMaterial
          color={colorObj}
          transparent
          opacity={0.3 + volume * 0.5}
          side={THREE.DoubleSide}
        />
      </mesh>
    </group>
  );
}
```
