### 问题 2：<bufferAttribute> 类型错误

**症状**：Starfield.tsx 和 ParticleSystem.tsx 中 `<bufferAttribute>` 缺少 `args` prop

**原因**：React Three Fiber v8+ 的类型定义要求 `bufferAttribute` 必须通过 `args` 传入 `[array, itemSize]`

**修复方案**：

**Starfield.tsx 修复：**
```tsx
import { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';

export function Starfield({ count = 6000 }: { count?: number }) {
  const meshRef = useRef<THREE.Points>(null);

  const { positions, sizes } = useMemo(() => {
    const positions = new Float32Array(count * 3);
    const sizes = new Float32Array(count);

    for (let i = 0; i < count; i++) {
      // 球形分布
      const radius = 50 + Math.random() * 100;
      const theta = Math.random() * Math.PI * 2;
      const phi = Math.acos(2 * Math.random() - 1);

      positions[i * 3] = radius * Math.sin(phi) * Math.cos(theta);
      positions[i * 3 + 1] = radius * Math.sin(phi) * Math.sin(theta);
      positions[i * 3 + 2] = radius * Math.cos(phi);

      sizes[i] = Math.random() * 2;
    }

    return { positions, sizes };
  }, [count]);

  useFrame((_, delta) => {
    if (meshRef.current) {
      meshRef.current.rotation.y += delta * 0.02;
      meshRef.current.rotation.x += delta * 0.01;
    }
  });

  return (
    <points ref={meshRef}>
      <bufferGeometry>
        {/* ✅ 正确：使用 args={[array, itemSize]} */}
        <bufferAttribute
          attach="attributes-position"
          args={[positions, 3]}
        />
        <bufferAttribute
          attach="attributes-size"
          args={[sizes, 1]}
        />
      </bufferGeometry>
      <pointsMaterial
        size={0.5}
        color="#e2e8f0"
        transparent
        opacity={0.8}
        sizeAttenuation
        depthWrite={false}
      />
    </points>
  );
}
```

**ParticleSystem.tsx 修复：**
```tsx
import { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';

interface ParticleSystemProps {
  particleCount?: number;
  color?: string;
}

export function ParticleSystem({ 
  particleCount = 1000,
  color = '#FFD700' 
}: ParticleSystemProps) {
  const meshRef = useRef<THREE.Points>(null);

  const { positions, velocities, colors } = useMemo(() => {
    const positions = new Float32Array(particleCount * 3);
    const velocities = new Float32Array(particleCount * 3);
    const colors = new Float32Array(particleCount * 3);

    const colorObj = new THREE.Color(color);

    for (let i = 0; i < particleCount; i++) {
      // 随机位置
      positions[i * 3] = (Math.random() - 0.5) * 50;
      positions[i * 3 + 1] = (Math.random() - 0.5) * 50;
      positions[i * 3 + 2] = (Math.random() - 0.5) * 50;

      // 随机速度
      velocities[i * 3] = (Math.random() - 0.5) * 0.1;
      velocities[i * 3 + 1] = (Math.random() - 0.5) * 0.1;
      velocities[i * 3 + 2] = (Math.random() - 0.5) * 0.1;

      // 颜色（带随机变化）
      colors[i * 3] = colorObj.r + (Math.random() - 0.5) * 0.2;
      colors[i * 3 + 1] = colorObj.g + (Math.random() - 0.5) * 0.2;
      colors[i * 3 + 2] = colorObj.b + (Math.random() - 0.5) * 0.2;
    }

    return { positions, velocities, colors };
  }, [particleCount, color]);

  useFrame((_, delta) => {
    if (!meshRef.current) return;

    const positionAttr = meshRef.current.geometry.attributes.position;
    const posArray = positionAttr.array as Float32Array;

    for (let i = 0; i < particleCount; i++) {
      posArray[i * 3] += velocities[i * 3] * delta;
      posArray[i * 3 + 1] += velocities[i * 3 + 1] * delta;
      posArray[i * 3 + 2] += velocities[i * 3 + 2] * delta;

      // 边界循环
      if (Math.abs(posArray[i * 3]) > 25) velocities[i * 3] *= -1;
      if (Math.abs(posArray[i * 3 + 1]) > 25) velocities[i * 3 + 1] *= -1;
      if (Math.abs(posArray[i * 3 + 2]) > 25) velocities[i * 3 + 2] *= -1;
    }

    positionAttr.needsUpdate = true;
  });

  return (
    <points ref={meshRef}>
      <bufferGeometry>
        {/* ✅ 正确：所有 bufferAttribute 都用 args */}
        <bufferAttribute
          attach="attributes-position"
          args={[positions, 3]}
        />
        <bufferAttribute
          attach="attributes-color"
          args={[colors, 3]}
        />
      </bufferGeometry>
      <pointsMaterial
        size={0.3}
        vertexColors
        transparent
        opacity={0.6}
        blending={THREE.AdditiveBlending}
        depthWrite={false}
      />
    </points>
  );
}
```

**关键规则**：R3F 中所有 `<bufferAttribute>` 必须这样写：
```tsx
// ❌ 错误
<bufferAttribute
  attach="attributes-position"
  count={positions.length / 3}
  array={positions}
  itemSize={3}
/>

// ✅ 正确
<bufferAttribute
  attach="attributes-position"
  args={[positions, 3]}  // [Float32Array, itemSize]
/>
```

### 问题 3：未使用变量 recorderError

**症状**：useVoiceChat.ts 第 39 行 `recorderError` 未被使用

**修复方案**（推荐方案 A，增加错误处理）：

```typescript
// src/voice/hooks/useVoiceChat.ts

import { useState, useEffect, useCallback, useRef } from 'react';
import { eventBus } from '@/utils/eventBus';

interface UseVoiceChatOptions {
  sessionId: string;
  onMessageReceived?: (message: any) => void;
}

export function useVoiceChat({ sessionId, onMessageReceived }: UseVoiceChatOptions) {
  const [messages, setMessages] = useState<any[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [recorderError, setRecorderError] = useState<string | null>(null);
  const wsRef = useRef<WebSocket | null>(null);

  // ✅ 使用 recorderError：监听错误变化并上报
  useEffect(() => {
    if (recorderError) {
      console.error('[VoiceChat] Recorder error:', recorderError);

      // 发送错误事件给全局错误处理
      eventBus.emit('app:error', {
        message: `录音出错: ${recorderError}`,
        code: 'RECORDER_ERROR',
        source: 'useVoiceChat',
      });

      // 3秒后自动清除错误
      const timer = setTimeout(() => setRecorderError(null), 3000);
      return () => clearTimeout(timer);
    }
  }, [recorderError]);

  const sendMessage = useCallback(async (content: string, isVoice: boolean = false) => {
    setIsLoading(true);
    setRecorderError(null); // 清除之前的错误

    try {
      // 添加用户消息到列表
      const userMessage = {
        id: Date.now().toString(),
        role: 'user' as const,
        content,
        timestamp: new Date().toISOString(),
        isVoice,
      };
      setMessages(prev => [...prev, userMessage]);

      // TODO: 发送给后端 API
      // const response = await fetch(`/api/v1/sessions/${sessionId}/messages`, {...});

      // 模拟 AI 回复（后续替换为真实 API）
      setTimeout(() => {
        const aiMessage = {
          id: (Date.now() + 1).toString(),
          role: 'assistant' as const,
          content: `收到您的消息：${content}`,
          timestamp: new Date().toISOString(),
          isVoice: false,
        };
        setMessages(prev => [...prev, aiMessage]);
        setIsLoading(false);
        onMessageReceived?.(aiMessage);
      }, 1000);

    } catch (error) {
      setRecorderError(error instanceof Error ? error.message : '未知错误');
      setIsLoading(false);
    }
  }, [sessionId, onMessageReceived]);

  return {
    messages,
    isLoading,
    recorderError, // 导出给 UI 显示
    sendMessage,
  };
}
```
